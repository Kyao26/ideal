=========================================
ALIS (BUY)
=========================================
a:=Ref(Abs(RMI(C,20,5)-50),-1);
coeff:=1.5+0.5*(1-a/50);
AP:=14;
hl2:=(H+L)/2;
atrv:=ATR(AP);
upBnd:=hl2-coeff*atrv;
dnBnd:=hl2+coeff*atrv;
fUp:=If(upBnd>PREV OR Ref(C,-1)<PREV,upBnd,PREV);
fDn:=If(dnBnd<PREV OR Ref(C,-1)>PREV,dnBnd,PREV);
dir:=If(Cum(1)<=AP,-1,If(PREV>0 AND C<fUp,-1,If(PREV<0 AND C>fDn,1,PREV)));
ST:=If(dir>0,fUp,fDn);
T:=HOUR()*60+MINUTE();
TF:=(T>=585 AND T<=1075);
WP:=If(W>0,W,(H+L+C)/3);
YGUN:=DAYOFMONTH()<>Ref(DAYOFMONTH(),-1);
GSAY:=Cum(If(YGUN,1,0));
KAY:=ValueWhen(1,YGUN,Ref(C,-1));
XD:=WP-KAY;
d:=ValueWhen(1,YGUN,Ref(Cum(XD*V),-1));
f:=ValueWhen(1,YGUN,Ref(Cum(V),-1));
g:=ValueWhen(1,YGUN,Ref(Cum(XD*XD*V),-1));
b:=ValueWhen(1,YGUN,Ref(Cum(1),-1));
DHAC:=Cum(V)-f;
ORTX:=If(DHAC>0,(Cum(XD*V)-d)/DHAC,0);
vwap:=If(DHAC>0,KAY+ORTX,C);
VARY:=If(DHAC>0,(Cum(XD*XD*V)-g)/DHAC-ORTX*ORTX,0);
sap:=Sqr(MAX(VARY,0));
barNo:=Cum(1)-b;
SOK:=GSAY>=2 AND barNo>=6 AND DHAC>0;
LFIL:=SOK AND C>vwap;
Cross(C,ST) AND TF AND LFIL
