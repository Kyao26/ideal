{ ==========================================================================
  OTT-A | SISTEM (System Tester) ve TARAMA (Explorer) KOSULLARI
  Tasarim 1 (adaptif opt) + Tasarim 3 (rejim anahtari)
  --------------------------------------------------------------------------
  REJIM ANAHTARI: sistem yonunu grafigin karakterine gore tayin eder.
      Hrs > 0.5 + s  -> kalici/trendli  -> OTT yonu uygulanir
      Hrs < 0.5 - s  -> salinimli       -> YON ALINMAZ (flat)
  Buradaki 0.5 keyfi bir esik degil, rastgele yuruyusun tam kendisidir:
  trend takibinin beklenen degeri ancak seri kaliciysa pozitiftir.
  s ise tahmincinin BIR STANDART HATASIDIR: s = Stdev(Hrs,20)/Sqrt(20).
  Sinir civarinda mod atmasini engeller; ayarlanabilir bir sayi degildir.

  PREV HAKKINDA (onemli)
  Asagidaki kosullar kendi kendine yeterlidir: OTT ratchet'i "Ott:=" degiskenine
  atanir ve PREV o degiskenin onceki degerine baglanir (MetaStock uyumlu
  davranis). Surumunuzde PREV kosulun kendi gecmisine baglanirsa ratchet bozulur;
  bu durumda OTT_Adaptif_v1'i kayitli indikator olarak olusturup System Tester
  icinden o indikatoru cagirin.
  ========================================================================== }


{ ================== AL KOSULU ============================================= }

per:=2;
N1:=(HHV(H,10)-LLV(L,10))/10;
N2:=(Ref(HHV(H,10),-10)-Ref(LLV(L,10),-10))/10;
N3:=(HHV(H,20)-LLV(L,20))/20;
Dfr:=If(N1>0 AND N2>0 AND N3>0,(Log(N1+N2)-Log(N3))/Log(2),1.5);
Hrs:=2-Dfr;
Hrs:=If(Hrs<0.2,0.2,If(Hrs>0.9,0.9,Hrs));
s:=Stdev(Hrs,20)/Sqrt(20);

C1:=Mov(C,per,VAR);
Rz:=(C-C1)/C;
SgR:=Stdev(Rz,20);
optA:=100*SgR/Hrs;
optA:=If(optA<0.05,0.05,If(optA>10,10,optA));

q1:=Ref(C1,-2)*(1+optA/200);
q2:=Ref(C1,-2)*(1-optA/200);
Ott:=If(Cum(1)=1,C1,If(q1<=PREV,q1,If(q2>=PREV,q2,PREV)));

Cross(C1,Ott) AND Hrs>0.5+s


{ ================== SAT KOSULU ============================================ }

per:=2;
N1:=(HHV(H,10)-LLV(L,10))/10;
N2:=(Ref(HHV(H,10),-10)-Ref(LLV(L,10),-10))/10;
N3:=(HHV(H,20)-LLV(L,20))/20;
Dfr:=If(N1>0 AND N2>0 AND N3>0,(Log(N1+N2)-Log(N3))/Log(2),1.5);
Hrs:=2-Dfr;
Hrs:=If(Hrs<0.2,0.2,If(Hrs>0.9,0.9,Hrs));
s:=Stdev(Hrs,20)/Sqrt(20);

C1:=Mov(C,per,VAR);
Rz:=(C-C1)/C;
SgR:=Stdev(Rz,20);
optA:=100*SgR/Hrs;
optA:=If(optA<0.05,0.05,If(optA>10,10,optA));

q1:=Ref(C1,-2)*(1+optA/200);
q2:=Ref(C1,-2)*(1-optA/200);
Ott:=If(Cum(1)=1,C1,If(q1<=PREV,q1,If(q2>=PREV,q2,PREV)));

Cross(Ott,C1) OR Hrs<0.5-s


{ ================== EXPLORER / TARAMA ====================================
  Bugun AL sinyali veren ve rejimi trendli olan semboller.
  Kolonlara ayrica Hrs ve optA ekleyin: tarama sonucunda hangi hissenin
  hangi karakterde oldugunu gorursunuz.
  ======================================================================== }

{ FILTRE kolonu: yukaridaki AL kosulunun aynisi.
  KOLON 1 : Hrs     -> rejim
  KOLON 2 : optA    -> o sembol icin sistemin sectigi bant yuzdesi
  KOLON 3 : Ott     -> takip eden stop seviyesi }
